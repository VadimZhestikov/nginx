● Here are real-world use cases that static nginx.conf cannot handle — grouped
  by which plan phase enables them:

  ---
  Phase 1 — Config introspection (read-only COM)

  Use case 1: Config validation & auditing

  Static conf gives no way to cross-validate blocks against each other. JS can:

  // nginx.conf
  js_include /etc/nginx/validate.js;

  // validate.js
  for (const server of nginx.http.servers) {
      // Enforce policy: every server must have a non-root `root`
      for (const loc of server.locations) {
          if (loc.root === "/") {
              nginx.log(NGX_LOG_ERR,
                  `POLICY VIOLATION: server "${server.name}" location 
  "${loc.path}" serves /`);
              // throw new Error("Invalid config") → aborts startup
          }
      }

      // Enforce: no server without a listen port (catch config mistakes)
      if (server.listen_ports.length === 0) {
          throw new Error(`server "${server.name}" has no listen directive`);
      }
  }
  nginx.log(NGX_LOG_NOTICE, `Config validated: ${nginx.http.servers.length} 
  servers OK`);

  Static alternative: none. You can't express cross-block validation in
  nginx.conf.

  ---
  Phase 2 — Config mutation at startup

  Use case 2: Environment-aware configuration

  The same nginx.conf runs in dev/staging/prod, with JS branching on a file or
  env variable:

  // js_include /etc/nginx/env-config.js

  const env = readFile("/etc/nginx/environment");  // "production" | "staging" |
   "dev"

  if (env.trim() === "production") {
      // Production: all servers point to production backends
      for (const ups of nginx.http.upstreams) {
          ups.peers[0].weight = 10;
          ups.peers[0].max_fails = 1;  // fail fast in prod
      }
      nginx.cycle.workers = nginx.cpu_count;  // max workers in prod
  } else {
      // Dev: single worker, verbose logging
      nginx.cycle.workers = 1;
  }

  Static alternative: requires separate nginx.conf files per environment,
  managed externally (Ansible, Helm, etc.). Any mistake means divergence between
   environments.

  ---
  Use case 3: Programmatic multi-tenant routing

  A SaaS product with 500 tenants, each with their own subdomain and document
  root:

  // js_include /etc/nginx/tenants.js

  const tenants = JSON.parse(readFile("/etc/nginx/tenants.json"));
  // [{ name: "acme", root: "/var/www/acme", backend: "10.0.1.5:3000" }, ...]

  const templateServer = nginx.http.servers[0];  // base config as template

  for (const tenant of tenants) {
      const server = nginx.http.addServer({
          server_name: [`${tenant.name}.example.com`],
          listen: [443],
      });
      server.root = tenant.root;
      server.addLocation("/api/", { proxy_pass: `http://${tenant.backend}` });
      server.addLocation("/",     { root: tenant.root });
  }

  // Remove the template server itself
  nginx.http.removeServer(templateServer);

  Static alternative: you must either write 500 server {} blocks manually or use
   a templating tool (Jinja2, etc.) that runs separately and produces a massive
  generated file. Any tenant onboarding requires re-running the template tool
  and reloading nginx.

  ---
  Use case 4: Time-based configuration

  // js_include /etc/nginx/time-config.js

  const hour = new Date().getHours();
  const isOffPeak = hour < 8 || hour > 20;

  // During off-peak hours, all traffic goes to the minimal cluster
  const upstream = nginx.http.upstreams.find(u => u.name === "backend");
  if (isOffPeak) {
      // Keep only the first peer (primary node), mark others as 
  backup-equivalent
      for (let i = 1; i < upstream.peers.length; i++) {
          upstream.peers[i].weight = 0;
      }
      nginx.cycle.workers = 2;
  } else {
      nginx.cycle.workers = nginx.cpu_count;
  }

  Static: nginx has no time-awareness in config at all. You'd need external cron
   + reload, which causes a process restart gap.

  ---
  Use case 5: Config from service discovery at boot

  // js_include /etc/nginx/consul.js

  // Read from a Consul snapshot file written by a sidecar process
  const services = JSON.parse(readFile("/run/consul-snapshot.json"));
  const upstream = nginx.http.upstreams.find(u => u.name === "app_backend");

  // Clear the static peers defined in nginx.conf and replace with discovered 
  ones
  for (const peer of [...upstream.peers]) {
      upstream.removePeer(peer.address);
  }
  for (const svc of services["app_backend"] ?? []) {
      upstream.addPeer(`${svc.address}:${svc.port}`, { weight: svc.weight ?? 1
  });
  }

  nginx.log(NGX_LOG_NOTICE,
      `Loaded ${services["app_backend"].length} peers from service discovery`);

  Static: you can't drive upstream peers from a file without custom C modules.
  The closest is ngx_http_upstream_zone_module + an external agent that calls
  the NGINX Plus API — a paid feature.

  ---
  Phase 3 — Runtime hot-reload (no restart)

  Use case 6: Blue/green deployment — instant zero-downtime switch

  // Called from a management API endpoint (Phase 4 handler)
  function switchToGreen(req) {
      const upstream = nginx.http.upstreams.find(u => u.name === "app");

      for (const peer of upstream.peers) {
          if (peer.tags.includes("blue"))  peer.down = true;   // drain blue
          if (peer.tags.includes("green")) peer.down = false;  // bring green up
      }

      req.respond(200, {}, "Switched to green");
  }

  Static alternative: requires nginx -s reload which forks new worker processes.
   During that ~100ms window, in-flight requests to blue are reset. The JS
  approach atomically flips a flag in shared memory — all workers see it
  instantly, no new processes are forked, zero dropped connections.

  ---
  Use case 7: Canary deployments with precise traffic control

  // Shift 10% of traffic to the canary on each call (ramp-up)
  function rampCanary(req) {
      const upstream = nginx.http.upstreams.find(u => u.name === "app");
      const canary = upstream.peers.find(p => p.tags.includes("canary"));
      const stable = upstream.peers.filter(p => !p.tags.includes("canary"));

      // Increase canary weight by 1, decrease stable proportionally
      canary.weight = Math.min(canary.weight + 1, 10);
      const stableWeight = 10 - canary.weight;
      for (const p of stable) p.weight = stableWeight / stable.length;

      req.respond(200, {}, JSON.stringify({
          canary_weight: canary.weight,
          stable_weight: stableWeight
      }));
  }

  Static: not possible without NGINX Plus paid upstream API or custom C modules.

  ---
  Use case 8: Automatic circuit breaking

  // Worker-side (Phase 4): track error rate per upstream peer in shared memory
  // When a peer error rate exceeds threshold, mark it down until it recovers
  function onRequestComplete(req, peer, elapsed_ms, status_code) {
      const upstream = nginx.http.upstreams.find(u => u.name === "app");
      const peer_obj = upstream.peers.find(p => p.address === peer);

      // Update rolling window stats (stored in JS shared state / nginx vars)
      peer_obj._errors = (peer_obj._errors ?? 0) + (status_code >= 500 ? 1 : 0);
      peer_obj._requests = (peer_obj._requests ?? 0) + 1;

      const errorRate = peer_obj._errors / peer_obj._requests;
      if (peer_obj._requests > 100 && errorRate > 0.5) {
          peer_obj.down = true;  // ← writes atomically to shared memory
          nginx.log(NGX_LOG_WARN,
              `Circuit open for ${peer}: ${(errorRate*100).toFixed(1)}% 
  errors`);
      }
  }

  Static nginx: has max_fails / fail_timeout per peer, but it is a fixed
  sliding-window count — you cannot apply custom logic (e.g., "fail only on 503,
   not 429", or "consider latency > 2s as a failure").

  ---
  Phase 4 — Request-phase JS handlers

  Use case 9: Request transformation impossible in static config

  NGINX's map, sub_filter, headers_more modules cover simple cases. JS covers
  the rest:

  // Route to different upstreams based on JWT claims (static nginx can only do 
  header existence checks)
  function routeByJWT(req) {
      const auth = req.headers["Authorization"] ?? "";
      const token = auth.replace("Bearer ", "");

      // Decode payload (base64url, no signature verification here — done 
  upstream)
      const payload = JSON.parse(atob(token.split(".")[1]));
      const tier = payload.tier;  // "free" | "pro" | "enterprise"

      // Rewrite the upstream based on tier
      const backendMap = {
          free:       "http://free-backend",
          pro:        "http://pro-backend",
          enterprise: "http://enterprise-backend",
      };

      // Set a variable that the proxy_pass directive reads
      req.variables["upstream_target"] = backendMap[tier] ?? backendMap.free;
      req.next();  // continue to proxy_pass $upstream_target
  }

  Static: nginx can check JWT header existence via map, but cannot parse and
  branch on JWT payload fields.

  ---
  Use case 10: Dynamic API gateway — route from a rules file

  // js_include /etc/nginx/api-gateway.js

  const routes = JSON.parse(readFile("/etc/nginx/routes.json"));
  // [{ prefix: "/v1/users", backend: "users-svc:3001", rateLimit: 100 },
  //  { prefix: "/v1/orders", backend: "orders-svc:3002", rateLimit: 50 }]

  // Build locations programmatically from the routes file
  const server = nginx.http.servers.find(s => s.name === "api.example.com");
  for (const route of routes) {
      server.addLocation(route.prefix, {
          proxy_pass: `http://${route.backend}`,
          rate_limit: route.rateLimit,
      });
  }

  vs. static equivalent which requires re-running a template tool on every route
   change.
